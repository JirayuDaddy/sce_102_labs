#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <time.h>

#define SIZE 3
#define MAX_PIECES 3
#define MAX_ROUNDS 3
#define WIN_THRESHOLD 2
#define MAX_NAME_LEN 50

// 1. Player Data Structure
typedef struct {
    char name[MAX_NAME_LEN];
    char symbol;     // 'X' or 'O'
    int score;        // Total overall score
    int rounds_won;   // Number of rounds won in this match
    int is_bot;       // 1 if Bot, 0 if Human
} 
Player;

// Global variables for game board and move history tracking
char board[SIZE][SIZE];
int moveCount[2] = {0, 0};
int moveRow[2][MAX_PIECES];
int moveCol[2][MAX_PIECES];

// Function Prototypes
void clearBuffer();
void initializeBoard();
void displayBoard();
int getPlayerIndex(char player);
int makeMove(char player, int row, int col);
int checkWin(char player);

void getBotMove(int *row, int *col);
void playGame(Player p1, Player p2);

// Clear stdin input buffer
void clearBuffer() {
    int c;
    while ((c = getchar()) != '\n' && c != EOF);
}

// Reset the board with empty spaces
void initializeBoard(){
    for (int row = 0; row < SIZE; row++)
{
        for (int col = 0; col < SIZE; col++) { board[row][col] = ' ';
        }
    }
    moveCount[0] = 0;
    moveCount[1] = 0;
}

// Display the current state of the board
void displayBoard() {
    printf("\n   1   2   3\n");
    for (int row = 0; row < SIZE; row++)
{
        printf("%d  ", row + 1);
        for (int col = 0; col < SIZE; col++) 
{
            printf("%c", board[row][col]);
            if (col < SIZE - 1) printf(" | ");
        }
        printf("\n");
        if (row < SIZE - 1) printf("  ---+---+---\n");
    }
    printf("\n");
}

int getPlayerIndex(char player) {
    return (player == 'X') ? 0 : 1;
}

// Handles placing pieces and removing the oldest piece if count > 3
int makeMove(char player, int row, int col) 
{
    if (row < 0 || row >= SIZE || col < 0 || col >= SIZE) {
        printf(" Invalid position! Please enter numbers between 1 and 3.\n");
        return 0;
    }

    if (board[row][col] != ' ') {
        printf(" That cell is already occupied!\n");

        return 0;
    }

    int playerIndex = getPlayerIndex(player);
    int *count = &moveCount[playerIndex];

    // If player already has 3 pieces on board, remove the oldest one
    if (*count == MAX_PIECES) {
        int oldRow = moveRow[playerIndex][0];
        int oldCol = moveCol[playerIndex][0];
        board[oldRow][oldCol] = ' ';

        // Shift move history array to the left
        for (int i = 0; i < MAX_PIECES - 1; i++) {
            moveRow[playerIndex][i] = moveRow[playerIndex][i + 1];
            moveCol[playerIndex][i] = moveCol[playerIndex][i + 1];
        }
        (*count)--;
    }

    // Place new piece on board
    board[row][col] = player;
    moveRow[playerIndex][*count] = row;
    moveCol[playerIndex][*count] = col;
    (*count)++;

    return 1;
}

// Check Win Logic
int checkWin(char player) {
    // Check rows and columns
    for (int i = 0; i < SIZE; i++) {
        if ((board[i][0] == player && board[i][1] == player && board[i][2] == player) ||
            (board[0][i] == player && board[1][i] == player && board[2][i] == player)) {
            return 1;
        }
    }
    // Check diagonals
    if ((board[0][0] == player && board[1][1] == player && board[2][2] == player) ||
        (board[0][2] == player && board[1][1] == player && board[2][0] == player)) {
        return 1;
    }
    return 0;
}

// Randomly select an empty space for Bot
void getBotMove(int *row, int *col) {
    int emptyCount = 0;
    int emptyRows[SIZE * SIZE];
    int emptyCols[SIZE * SIZE];

    for (int r = 0; r < SIZE; r++) {
        for (int c = 0; c < SIZE; c++) {
            if (board[r][c] == ' ') {
                emptyRows[emptyCount] = r;
                emptyCols[emptyCount] = c;
                emptyCount++;
            }
        }
    }

    if (emptyCount > 0) {
        int randomIndex = rand() % emptyCount;
        *row = emptyRows[randomIndex];
        *col = emptyCols[randomIndex];
    }
}

// Main Game Engine Loop
void playGame(Player p1, Player p2) {
    printf("\n==================================================\n");
    printf("  Welcome to Memory Tic-Tac-Toe (Best of 3 Rounds)\n");
    printf("  * Special Rule: Max 3 pieces per player on board *\n");
    printf("==================================================\n");

    int current_round = 1;

    // Best of 3 Match Loop
    while (p1.rounds_won < WIN_THRESHOLD && p2.rounds_won < WIN_THRESHOLD && current_round <= MAX_ROUNDS) {
        initializeBoard();
        Player *turn = &p1; // Start with Player 1 (X)
        int round_over = 0;

        printf("\n==========================================\n");
        printf("                 Round %d\n", current_round);
        printf("==========================================\n");

        while (!round_over) {
            displayBoard();
            int row, col;

            if (turn->is_bot) {
                printf("%s (%c) is thinking...\n", turn->name, turn->symbol);
                getBotMove(&row, &col);
                printf("%s picked row %d, col %d\n", turn->name, row + 1, col + 1);
            } else {
                printf("%s (%c), it's your turn.\n", turn->name, turn->symbol);
                printf("Enter row and column (1-3) e.g. '1 2' (or enter '0 0' to quit): ");

                if (scanf("%d %d", &row, &col) != 2) {
                    printf(" Invalid input!\n");
                    clearBuffer();
                    continue;
                }

                if (row == 0 && col == 0) {
                    printf("\nMatch terminated by player.\n");
                    return;
                }

                // Convert 1-3 input to 0-2 array indices
                row--;
                col--;
            }

            // Make the move
            if (makeMove(turn->symbol, row, col)) {
                // Check win condition
                if (checkWin(turn->symbol)) {
                    displayBoard();
                    printf(" %s won Round %d!\n", turn->name, current_round);
                    
                    // Update scores and rounds won
                    turn->rounds_won++;
                    turn->score += 100;
                    round_over = 1;
                } else {
                    // Switch turn to the other player
                    turn = (turn->symbol == p1.symbol) ? &p2 : &p1;
                }
            }
        }

        // Display match status after each round
        printf("\n Score after Round %d:\n", current_round);
        printf("   %s: %d round(s) | %s: %d round(s)\n", p1.name, p1.rounds_won, p2.name, p2.rounds_won);

        current_round++;
    }

    // Final match summary
    printf("\n==========================================\n");
    printf("           Final Match Summary            \n");
    printf("==========================================\n");
    
    if (p1.rounds_won > p2.rounds_won) {
        printf(" Match Winner: %s!\n", p1.name);
    } else if (p2.rounds_won > p1.rounds_won) {
        printf(" Match Winner: %s!\n", p2.name);
    } else {
        printf(" Match Result: It's a draw!\n");
    }

    printf("Total Scores -> %s: %d pts | %s: %d pts\n", p1.name, p1.score, p2.name, p2.score);
    printf("==========================================\n");
}

int main() {
    srand(time(NULL)); // Seed for Bot's random moves

    int menu;
    Player p1, p2;

    // Default Initialization
    p1.symbol = 'X';
    p1.score = 0;
    p1.rounds_won = 0;
    p1.is_bot = 0;

    p2.symbol = 'O';
    p2.score = 0;
    p2.rounds_won = 0;
    p2.is_bot = 0;

    printf("\n");
    printf("================================\n");
    printf("            XO Game             \n");
    printf("================================\n");
    printf("  [1] Play with Bot\n");
    printf("  [2] Play with Friend\n");
    printf("  [0] Exit Game\n");
    printf("================================\n");
    printf("  Select Menu: ");

    if (scanf("%d", &menu) != 1) {
        printf("Invalid input!\n");
        return 1;
    }
    clearBuffer();

    switch (menu) {
        case 1:
            printf("\nYou selected Play with Bot!\n");
            printf("Enter your name: ");
            fgets(p1.name, MAX_NAME_LEN, stdin);
            p1.name[strcspn(p1.name, "\n")] = 0;

            strcpy(p2.name, "Bot");
            p2.is_bot = 1;

            printf("\n--- Game Started: %s VS %s ---\n", p1.name, p2.name);
            playGame(p1, p2);
            break;

        case 2:
            printf("\nYou selected Play with Friend!\n");
            
            printf("Enter Player 1 name: ");
            fgets(p1.name, MAX_NAME_LEN, stdin);
            p1.name[strcspn(p1.name, "\n")] = 0;

            printf("Enter Player 2 name: ");
            fgets(p2.name, MAX_NAME_LEN, stdin);
            p2.name[strcspn(p2.name, "\n")] = 0;

            printf("\n--- Game Started: %s VS %s ---\n", p1.name, p2.name);
            playGame(p1, p2);
            break;

        case 0:
            printf("\nThank you for playing!\n");
            printf("Exiting game...\n");
            break;

        default:
            printf("\nInvalid menu! Please try again.\n");
            break;
    }

    return 0;
}
